Листинг исходного кода программы

В данном приложении представлен полный листинг исходного кода разработанной автоматизированной системы управления компьютерным клубом «ModeLink». Код структурирован по модулям в соответствии с архитектурой MVC (Model-View-Controller) и включает в себя:

  • Определение моделей данных (ORM);
  • Сервисный слой бизнес-логики;
  • Контроллеры API и обработку запросов;
  • Логику Telegram-бота на платформе Node.js.

class Tariff(models.Model):

    name = models.CharField(max_length=100, verbose_name="Название тарифа")
    hourly_rate = models.DecimalField(max_digits=10, decimal_places=2, verbose_name="Цена за час (₽)")
    description = models.TextField(blank=True, verbose_name="Описание")
    is_active = models.BooleanField(default=True, verbose_name="Активен")

def send_telegram_notification(client_id, message):
    try:
        from loyalty.models import Client
        client = Client.objects.get(pk=client_id)
        telegram_id = client.telegram_id
        if not telegram_id: return False

        token = getattr(settings, 'TELEGRAM_BOT_TOKEN', '')
        url = f"https://api.telegram.org/bot{token}/sendMessage"
        payload = {'chat_id': telegram_id, 'text': message, 'parse_mode': 'Markdown'}
        response = requests.post(url, json=payload, timeout=5)
        response.raise_for_status()
        return True
    except Exception as e:
        logger.error(f"Failed to send notification: {e}")
        return False

@transaction.atomic
def start_session(client, place, is_guest=False, guest_name=None, duration_minutes=None, promotion_id=None):
    place = Place.objects.select_for_update().get(pk=place.pk)
    if place.status != 'free':
        raise ValueError(f"Место {place.name} не свободно")

    start_time = timezone.now()
    scheduled_end_time = None
    if duration_minutes:
        scheduled_end_time = start_time + timedelta(minutes=int(duration_minutes))

    promotion = None
    discount_percentage = 0
    if promotion_id:
        try:
            promotion = Promotion.objects.get(pk=promotion_id, is_active=True)
            discount_percentage = promotion.discount_percentage
        except Promotion.DoesNotExist: pass

    session = Session.objects.create(
        client=client if not is_guest else None,
        place=place,
        start_time=start_time,
        scheduled_end_time=scheduled_end_time,
        status='active',
        is_guest=is_guest,
        guest_name=guest_name if is_guest else None,
        promotion=promotion,
        discount_applied=discount_percentage
    )
    place.status = 'occupied'
    place.save()
    return session

@transaction.atomic
def stop_session(session_id):
    try:
        session = Session.objects.select_for_update().get(pk=session_id)
    except Session.DoesNotExist:
        raise ValueError("Сессия не найдена")

    if session.status == 'completed':
        raise ValueError("Сессия уже завершена")

    session.end_time = timezone.now()
    duration = session.end_time - session.start_time
    hours_played_exact = Decimal(duration.total_seconds()) / Decimal(3600)
    hours_to_charge = Decimal(math.ceil(hours_played_exact)) or Decimal(1)

    hourly_rate = session.place.tariff.hourly_rate if session.place.tariff else Decimal(300)
    cost = hours_to_charge * hourly_rate
    
    if session.discount_applied > 0:
        cost = cost * (Decimal(1) - (Decimal(session.discount_applied) / Decimal(100)))

    session.cost = cost
    session.status = 'completed'
    session.place.status = 'free'
    session.place.save()
    session.save()

    bonus_points_added = 0
    if not session.is_guest and session.client:
        session.client.total_hours_played += hours_played_exact
        bonus_points_added = int(cost * Decimal(0.01))
        session.client.bonus_points += bonus_points_added
        session.client.save()
        send_telegram_notification(session.client.id, f"✅ Сессия завершена. Стоимость: {cost}₽")

    return {"session": session, "cost": cost, "bonus": bonus_points_added}

def get_place_pricing_recommendation(place_id):
    place = Place.objects.get(pk=place_id)
    now = timezone.now()
    three_hours_ago = now - timedelta(hours=3)
    
    sessions = Session.objects.filter(
        place=place, start_time__lt=now,
        end_time__gt=three_hours_ago
    )
    total_sec = 0
    for s in sessions:
        s_start = max(s.start_time, three_hours_ago)
        s_end = min(s.end_time or now, now)
        if s_end > s_start:
            total_sec += (s_end - s_start).total_seconds()
            
    rate = total_sec / (3 * 3600)
    res = {"place": place.name, "occupancy": round(rate * 100, 1)}
    if rate < 0.3: res["recommendation"] = "-15% (Low demand)"
    elif rate > 0.8: res["recommendation"] = "+10% (High demand)"
    else: res["recommendation"] = "Keep current"
    return res

@api_view(['POST'])
def start_session_api(request):
    telegram_id = request.data.get('telegram_id')
    place_id = request.data.get('place_id')
    client_name = request.data.get('client_name', '').strip()
    is_guest = request.data.get('is_guest', False)

    if not place_id: return Response({"error": "No place_id"}, 400)

    try:
        place = Place.objects.get(pk=place_id)
        if is_guest:
            session = start_session(None, place, is_guest=True, 
                                   guest_name=request.data.get('guest_name'))
            return Response({"session_id": session.id, "status": "started"}, 201)
        
        try:
            client = Client.objects.get(telegram_id=telegram_id)
        except Client.DoesNotExist:
            client = Client.objects.create(telegram_id=telegram_id, name=client_name, rank='Новичок')

        session = start_session(client, place, duration_minutes=request.data.get('duration'))
        return Response({"session_id": session.id, "client": client.name}, 201)
    except Exception as e:
        return Response({"error": str(e)}, 500)

@api_view(['GET'])
def manager_stats(request):
    from django.db.models import Sum, Avg
    now = timezone.now().replace(hour=0, minute=0, second=0)
    daily = Session.objects.filter(start_time__gte=now)
    
    total_rev = daily.filter(status='completed').aggregate(Sum('cost'))['cost__sum'] or 0
    avg_cost = daily.filter(status='completed').aggregate(Avg('cost'))['cost__avg'] or 0
    active_count = daily.filter(status='active').count()

    return Response({
        "revenue": float(total_rev),
        "avg_session": float(avg_cost),
        "active_sessions": active_count
    })

require('dotenv').config();
const TelegramBot = require('node-telegram-bot-api');
const axios = require('axios');

const token = process.env.BOT_TOKEN;
const API_BASE_URL = process.env.API_BASE_URL || 'http://127.0.0.1:8000/api';

const bot = new TelegramBot(token, { polling: true });
const userStates = {};

bot.onText(/\/start/, async (msg) => {
    const chatId = msg.chat.id;
    try {
        const client = await axios.get(`${API_BASE_URL}/client_info/${chatId}/`).then(r => r.data).catch(() => null);
        if (client) {
            bot.sendMessage(chatId, `С возвращением, ${client.name}!`, mainKeyboard());
        } else {
            userStates[chatId] = { step: 'WAIT_NAME' };
            bot.sendMessage(chatId, 'Добро пожаловать! Как вас зовут?');
        }
    } catch (e) { bot.sendMessage(chatId, 'Ошибка сервера.'); }
});

bot.on('message', async (msg) => {
    const chatId = msg.chat.id;
    const text = msg.text;
    if (!text || text.startsWith('/')) return;

    const state = userStates[chatId];
    if (state && state.step === 'WAIT_NAME') {
        await axios.post(`${API_BASE_URL}/register_client/`, { telegram_id: chatId, name: text });
        delete userStates[chatId];
        return bot.sendMessage(chatId, `Приятно познакомиться, ${text}!`, mainKeyboard());
    }

    if (text === '🎮 Статус клуба') {
        const places = await axios.get(`${API_BASE_URL}/place_status/`).then(r => r.data);
        let statusText = '🎮 Карта клуба:\n\n';
        places.forEach(p => {
            const icon = p.status === 'free' ? '🟢' : '🔴';
            statusText += `${icon} ${p.name} (${p.place_type})\n`;
        });
        bot.sendMessage(chatId, statusText, mainKeyboard());
    }
});

function mainKeyboard() {
    return {

            keyboard: [

                ['📅 Забронировать', '📋 Мои брони']
            ],
            resize_keyboard: true
        }
    };
}

        completed: { text: 'Завершено', color: '#64748b', icon: '✔️' },
